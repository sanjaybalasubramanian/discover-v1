"""
Email extraction from text with evidence
"""
import re
from typing import List, Optional


class ExtractedEmail:
    """Extracted email with context"""
    def __init__(self, email: str, surrounding_text: str, name: str = None, title: str = None):
        self.email = email
        self.surrounding_text = surrounding_text
        self.name = name
        self.title = title

    def to_dict(self):
        return {
            'email': self.email,
            'surrounding_text': self.surrounding_text,
            'name': self.name,
            'title': self.title
        }

def extract_emails(text: str, source_url: str) -> List[ExtractedEmail]:
    """
    Extract emails from text with surrounding context
    """
    if not text:
        return []

    # Normalize obfuscations
    normalized = text
    replacements = [
        ('[at]', '@'), ('(at)', '@'), (' at ', '@'),
        ('[dot]', '.'), ('(dot)', '.'), (' dot ', '.'),
        ('[AT]', '@'), ('(AT)', '@'),
        ('[DOT]', '.'), ('(DOT)', '.')
    ]

    for old, new in replacements:
        normalized = normalized.replace(old, new)

    # Email regex
    email_pattern = r'\b[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Z|a-z]{2,}\b'
    matches = re.finditer(email_pattern, normalized)

    extracted = []
    seen = set()

    for match in matches:
        email = match.group(0).lower().strip()

        # Skip if already seen
        if email in seen:
            continue
        seen.add(email)

        # Skip junk emails
        if is_junk_email(email):
            continue

        # Extract surrounding text (200 chars before and after)
        start = max(0, match.start() - 100)
        end = min(len(normalized), match.end() + 100)
        surrounding = normalized[start:end].strip()

        # Try to extract name and title
        name, title = extract_name_and_title(surrounding, email)

        extracted.append(ExtractedEmail(email, surrounding, name, title))

    return extracted

def is_junk_email(email: str) -> bool:
    """Check if email is junk/invalid"""
    junk_patterns = [
        'example.com', 'test.com', 'sentry.io', 'wixpress.com',
        'placeholder', 'noreply', 'no-reply', 'donotreply',
        'john.doe@', 'jane.doe@', 'john.smith@', 'jane.smith@',
        'johndoe@', 'janedoe@', 'rxxx@', 'xxxx@',
        'user@', 'email@', 'name@'
        # Removed: 'admin@', 'info@', 'support@', 'hello@', 'contact@'
        # Removed: 'careers@', 'hr@', 'recruiting@' - these are valid!
    ]

    for pattern in junk_patterns:
        if pattern in email.lower():
            return True

    # Skip emails with file extensions
    if re.search(r'\.(jpg|png|gif|svg|css|js|pdf)$', email, re.I):
        return True

    return False

def extract_name_and_title(text: str, email: str) -> tuple[Optional[str], Optional[str]]:
    """Extract name and title from surrounding text"""
    name = None
    title = None

    # Title keywords
    title_keywords = [
        'recruiter', 'talent acquisition', 'hr', 'human resources',
        'hiring', 'manager', 'lead', 'engineer', 'director', 'specialist'
    ]

    # Split into lines
    lines = re.split(r'\n|<br>|;', text)

    for line in lines:
        if email.lower() in line.lower():
            # Found line with email
            parts = re.split(r'[,\-â€“â€”|]', line)

            for part in parts:
                trimmed = part.strip()

                # Check for title keywords
                if not title:
                    for keyword in title_keywords:
                        if keyword in trimmed.lower():
                            title = trimmed[:100]  # Limit length
                            break

                # Check if looks like a name (2-3 capitalized words)
                if not name and re.match(r'^[A-Z][a-z]+ [A-Z][a-z]+', trimmed):
                    name = ' '.join(trimmed.split()[:3])

            break

    return name, title

def extract_from_snippet(snippet: str, url: str) -> List[ExtractedEmail]:
    """Extract emails from search result snippet"""
    return extract_emails(snippet, url)

def rank_contacts_by_relevance(extracted_emails: List[ExtractedEmail]) -> List[dict]:
    """
    Rank extracted emails by likelihood of being recruiter contacts.
    Simple heuristic-based approach - no LLM needed!

    Returns list of dicts with: email, name, title, confidence
    """
    ranked = []

    for extracted in extracted_emails:
        email = extracted.email.lower()
        context = extracted.surrounding_text.lower()
        score = 0.0

        # HIGH PRIORITY: Email prefix suggests recruiting role
        high_priority_prefixes = ['careers', 'jobs', 'recruit', 'talent', 'hr', 'hiring']
        if any(prefix in email for prefix in high_priority_prefixes):
            score += 0.9

        # MEDIUM PRIORITY: Context suggests recruiting role
        recruiting_keywords = [
            'recruiter', 'talent acquisition', 'hiring', 'hr',
            'human resources', 'people', 'talent', 'recruitment'
        ]
        if any(keyword in context for keyword in recruiting_keywords):
            score += 0.7

        # BONUS: Has title or name extracted
        if extracted.title:
            score += 0.1
        if extracted.name:
            score += 0.1

        # PENALTY: Generic contact emails (likely not personal recruiter)
        generic_prefixes = ['info', 'contact', 'hello', 'support', 'sales', 'press']
        if any(prefix in email for prefix in generic_prefixes):
            score = max(0.2, score - 0.6)  # Lower score but not zero

        # Cap at 1.0
        confidence = min(score, 1.0)

        ranked.append({
            'email': extracted.email,
            'name': extracted.name,
            'title': extracted.title,
            'confidence': confidence,
            'context': extracted.surrounding_text[:200]  # Keep some context
        })

    # Sort by confidence (highest first)
    return sorted(ranked, key=lambda x: x['confidence'], reverse=True)

"""
Web page fetching using Crawl4AI with domain restrictions
"""
import asyncio
from typing import Optional
from crawl4ai import AsyncWebCrawler
from urllib.parse import urlparse

# CRITICAL: Never fetch these domains directly - use snippets only
RESTRICTED_FETCH_DOMAINS = [
    'linkedin.com',
    'www.linkedin.com',
    'naukri.com',
    'www.naukri.com'
]

def should_skip_fetch(url: str) -> bool:
    """
    Check if URL should be skipped (LinkedIn, Naukri)
    These sites provide good snippets, don't need full fetch
    """
    try:
        domain = urlparse(url).netloc.lower()
        return any(domain.endswith(restricted) for restricted in RESTRICTED_FETCH_DOMAINS)
    except Exception:
        return False

async def fetch_page(url: str) -> Optional[str]:
    """
    Fetch full page content using Crawl4AI
    
    IMPORTANT: Skips LinkedIn and Naukri (use snippets for those)
    
    Args:
        url: Page URL to fetch
    
    Returns:
        Clean text content or None if skipped/failed
    """
    
    # Check if domain is restricted
    if should_skip_fetch(url):
        print(f"[Fetch] Skipping restricted domain: {url}")
        return None
    
    try:
        async with AsyncWebCrawler(verbose=False) as crawler:
            result = await crawler.arun(
                url=url,
                word_count_threshold=10,
                bypass_cache=False,
                screenshot=False
            )
            
            if result.success:
                # Return markdown text (cleaner than HTML)
                content = result.markdown or result.cleaned_html or ""
                print(f"[Fetch] Success: {url} ({len(content)} chars)")
                return content
            else:
                print(f"[Fetch] Failed: {url} - {result.error_message}")
                return None
                
    except Exception as e:
        print(f"[Fetch] Error: {url} - {e}")
        return None

async def fetch_pages_batch(urls: list[str], max_concurrent: int = 5) -> dict[str, Optional[str]]:
    """
    Fetch multiple pages concurrently (skips restricted domains)
    
    Args:
        urls: List of URLs to fetch
        max_concurrent: Maximum concurrent requests
    
    Returns:
        Dict mapping URL to content (or None if failed/skipped)
    """
    
    semaphore = asyncio.Semaphore(max_concurrent)
    
    async def fetch_with_limit(url: str):
        async with semaphore:
            content = await fetch_page(url)
            return url, content
    
    tasks = [fetch_with_limit(url) for url in urls]
    results = await asyncio.gather(*tasks)
    
    return dict(results)

"""
Job relevance checking - simple keyword-based approach
No LLM needed!
"""

def is_relevant_job(title: str, description: str) -> tuple[bool, float]:
    """
    Check if job is relevant for GenAI/LLM/AI Engineer role.
    Returns (is_relevant, confidence_score)

    Simple keyword matching approach - fast and free!
    """
    if not title and not description:
        return False, 0.0

    # Combine title and description for checking
    text = f"{title} {description}".lower()

    # HIGH PRIORITY KEYWORDS (strong signals)
    high_priority = [
        'genai', 'gen ai', 'generative ai',
        'llm', 'large language model',
        'ai engineer', 'ml engineer',
        'machine learning engineer',
        'deep learning engineer',
        'artificial intelligence engineer'
    ]

    # MEDIUM PRIORITY KEYWORDS (good signals)
    medium_priority = [
        'ai', 'artificial intelligence',
        'machine learning', 'ml',
        'deep learning', 'neural network',
        'nlp', 'natural language processing',
        'computer vision', 'cv',
        'pytorch', 'tensorflow', 'keras',
        'transformers', 'bert', 'gpt'
    ]

    # NEGATIVE KEYWORDS (likely not what we want)
    negative_keywords = [
        'data analyst', 'business analyst',
        'sales', 'marketing', 'customer success',
        'support engineer', 'qa engineer', 'test engineer',
        'devops engineer', 'sre', 'infrastructure',
        'frontend', 'full stack', 'web developer'
    ]

    score = 0.0

    # Check high priority keywords
    for keyword in high_priority:
        if keyword in text:
            score += 0.4

    # Check medium priority keywords
    for keyword in medium_priority:
        if keyword in text:
            score += 0.2

    # Check negative keywords (penalties)
    for keyword in negative_keywords:
        if keyword in text:
            score -= 0.3

    # Bonus for "engineer" in title
    if 'engineer' in title.lower():
        score += 0.1

    # Cap score between 0 and 1
    confidence = max(0.0, min(1.0, score))

    # Consider relevant if confidence >= 0.5
    is_relevant = confidence >= 0.5

    return is_relevant, confidence

"""
AWS Bedrock LLM client for intelligent parsing and extraction
Supports Claude 3.5 Sonnet and Claude 3 Haiku
"""
import json
import os
from typing import List, Optional

import boto3
from pydantic import BaseModel

# Initialize Bedrock client
bedrock = boto3.client(
    service_name='bedrock-runtime',
    region_name=os.getenv('AWS_REGION', 'us-east-1'),
    aws_access_key_id=os.getenv('AWS_ACCESS_KEY_ID'),
    aws_secret_access_key=os.getenv('AWS_SECRET_ACCESS_KEY')
)

# Model configuration
BEDROCK_MODEL = os.getenv('BEDROCK_MODEL', 'anthropic.claude-3-5-sonnet-20240620-v1:0')
# Options:
# - anthropic.claude-3-5-sonnet-20240620-v1:0 (smart, expensive)
# - anthropic.claude-3-haiku-20240307-v1:0 (fast, cheap)

# Structured outputs
class ParsedJob(BaseModel):
    """Structured job data"""
    company_name: str
    role_title: str
    location: str
    experience_years: Optional[int] = None
    is_relevant: bool
    confidence: float  # 0-1

class ExtractedContact(BaseModel):
    """Structured contact data"""
    email: str
    name: Optional[str] = None
    title: Optional[str] = None
    is_recruiter: bool
    confidence: float  # 0-1
    reasoning: str

async def invoke_bedrock(prompt: str, system: str = "", max_tokens: int = 500) -> str:
    """
    Call AWS Bedrock with Claude model

    Args:
        prompt: User prompt
        system: System message
        max_tokens: Max response tokens

    Returns:
        Model response text
    """

    body = {
        "anthropic_version": "bedrock-2023-05-31",
        "max_tokens": max_tokens,
        "thinking": {
            "type": "adaptive"
        },
        "output_config": {
            "effort": "low"
        },
        "messages": [
            {
                "role": "user",
                "content": prompt
            }
        ]
    }

    if system:
        body["system"] = system

    try:
        response = bedrock.invoke_model(
            modelId=BEDROCK_MODEL,
            body=json.dumps(body)
        )

        response_body = json.loads(response['body'].read())

        # Handle different response formats
        if 'content' in response_body:
            # Standard format: {"content": [{"text": "..."}, {"type": "thinking", ...}]}
            content_items = response_body['content']

            # Find the text content (skip thinking/other types)
            for item in content_items:
                if isinstance(item, dict):
                    if item.get('type') == 'text' and 'text' in item:
                        return item['text']
                    elif 'text' in item and item.get('type') != 'thinking':
                        return item['text']

            # Fallback: if no text found, check if there's thinking output
            for item in content_items:
                if isinstance(item, dict) and 'thinking' in item:
                    return item['thinking']

            # Last resort: return first item with text
            if content_items and isinstance(content_items[0], dict) and 'text' in content_items[0]:
                return content_items[0]['text']

            raise ValueError(f"No text content found in response: {content_items}")

        elif 'completion' in response_body:
            # Legacy format: {"completion": "..."}
            return response_body['completion']
        else:
            # Fallback: try to extract text from any structure
            print(f"Unknown response format: {response_body.keys()}")
            raise ValueError(f"Unexpected response format: {list(response_body.keys())}")

    except KeyError as e:
        print(f"Bedrock API error: Missing key '{e}' in response")
        print(f"Response structure: {response_body if 'response_body' in locals() else 'N/A'}")
        raise
    except Exception as e:
        print(f"Bedrock API error: {e}")
        raise

async def parse_job_listing(
    title: str,
    url: str,
    snippet: str,
    full_text: Optional[str] = None
) -> ParsedJob:
    """
    Parse job listing using LLM to extract structured data

    Args:
        title: Raw search result title
        url: Job listing URL
        snippet: Search result snippet
        full_text: Full page text (if available)

    Returns:
        ParsedJob with clean, structured data
    """

    context = full_text if full_text else snippet

    prompt = f"""You are an expert at parsing job listings. Extract structured data from this job posting.

URL: {url}
Title: {title}
Content: {context[:2000]}

Extract:
1. company_name: The company hiring (clean name only, no extra text)
2. role_title: The job role (clean title, e.g., "GenAI Engineer", "AI Engineer")
3. location: Primary work location (city name or "Remote")
4. experience_years: Required years of experience (integer, null if not specified)
5. is_relevant: Is this a GenAI/LLM/AI Engineer role? (boolean)
6. confidence: Your confidence in the extraction (0.0 to 1.0)

Return ONLY valid JSON:
{{
  "company_name": "string",
  "role_title": "string",
  "location": "string",
  "experience_years": number or null,
  "is_relevant": boolean,
  "confidence": number
}}

Rules:
- Clean company names (remove "Jobs at", "Careers", etc.)
- Standardize role titles (use "GenAI Engineer" not "Gen AI Eng")
- Location should be city name or "Remote" or "Remote (India)"
- Only mark is_relevant=true for AI/ML/GenAI engineering roles
"""

    system = "You are a precise job listing parser. Always return valid JSON only, no markdown or extra text."

    try:
        response_text = await invoke_bedrock(prompt, system, max_tokens=300)

        # Extract JSON from response (Claude sometimes adds markdown)
        json_start = response_text.find('{')
        json_end = response_text.rfind('}') + 1
        json_str = response_text[json_start:json_end]

        result = json.loads(json_str)
        return ParsedJob(**result)

    except Exception as e:
        print(f"LLM parsing failed: {e}")
        # Fallback to basic parsing
        return ParsedJob(
            company_name=title.split('|')[0].strip() if '|' in title else "Unknown",
            role_title=title.split('|')[1].strip() if '|' in title else title[:100],
            location="Unknown",
            experience_years=None,
            is_relevant=True,
            confidence=0.3
        )

async def extract_best_contacts(
    page_text: str,
    url: str,
    found_emails: List[str]
) -> List[ExtractedContact]:
    """
    Use LLM to identify best contacts from extracted emails

    Args:
        page_text: Full page content
        url: Source URL
        found_emails: List of email addresses found via regex

    Returns:
        List of ExtractedContact, ranked by relevance
    """

    if not found_emails:
        return []

    # Limit context to reduce tokens
    context_snippet = page_text[:3000]

    prompt = f"""You are an expert at identifying the best contact person for job applications.

Source: {url}
Context: {context_snippet}

Found these email addresses:
{json.dumps(found_emails)}

For EACH email, determine:
1. email: The email address
2. name: Person's name if mentioned (or null)
3. title: Person's job title if mentioned (or null)
4. is_recruiter: Is this a recruiter/HR/hiring manager? (boolean)
5. confidence: How confident are you this is a good contact? (0.0 to 1.0)
6. reasoning: Brief explanation (1 sentence)

Ranking: Named recruiters > HR emails > Generic emails
Avoid: noreply@, info@, admin@

Return valid JSON array sorted by confidence (highest first):
[{{"email": "...", "name": null, "title": null, "is_recruiter": true, "confidence": 0.9, "reasoning": "..."}}]
"""

    system = "You are an expert at evaluating job contacts. Return valid JSON array only."

    try:
        response_text = await invoke_bedrock(prompt, system, max_tokens=800)

        # Extract JSON array from response
        json_start = response_text.find('[')
        json_end = response_text.rfind(']') + 1
        json_str = response_text[json_start:json_end]

        contacts_data = json.loads(json_str)

        return [ExtractedContact(**contact) for contact in contacts_data]

    except Exception as e:
        print(f"LLM contact extraction failed: {e}")
        # Fallback
        return [
            ExtractedContact(
                email=email,
                name=None,
                title=None,
                is_recruiter='hr' in email.lower() or 'career' in email.lower(),
                confidence=0.5,
                reasoning="Fallback scoring"
            )
            for email in found_emails
        ]

'''Main pipeline runner - using JobSpy for job discovery'''
import os
from datetime import datetime
from typing import Any, Dict

from sqlalchemy.orm import Session

from app.config import calculate_fit_score, matches_location
from app.models import Company, Job, Run
from app.models.contact import Contact
from app.search.jobspy_scraper import JobSpyScraper, build_jobspy_searches


async def run_pipeline(run_id: int, db: Session) -> Dict[str, Any]:
    """Main outreach pipeline with JobSpy"""
    run = db.query(Run).filter(Run.id == run_id).first()
    if not run:
        raise Exception(f"Run {run_id} not found")
    try:
        print(f"[Pipeline] Starting run #{run_id} with JobSpy")
        scraper = JobSpyScraper()
        searches = build_jobspy_searches()
        jobs_found = 0
        contacts_found = 0
        processed_urls = set()
        for search_config in searches:
            print(f"[Pipeline] Scraping: {search_config['search_term']} in {search_config['location']}")
            job_results = scraper.scrape_jobs(**search_config)
            for job_data in job_results:
                job_url = job_data['job_url']
                if job_url in processed_urls:
                    continue
                processed_urls.add(job_url)

                # Validate required fields
                if not job_data.get('title'):
                    print("[Pipeline] ⚠️  Skipping - no job title")
                    continue

                if not job_data.get('company'):
                    print("[Pipeline] ⚠️  Skipping - no company name")
                    continue

                print(f"[Pipeline] Processing: {job_data['title'][:60]}...")

                # Get description (may be empty for some jobs)
                description = job_data.get('description', '')
                if not description:
                    print("[Pipeline] ⚠️  No description available, using title only for relevance check")

                # Check relevance with simple keyword matching (no LLM!)
                from app.filters.relevance import is_relevant_job
                is_relevant, confidence = is_relevant_job(
                    title=job_data['title'],
                    description=description
                )

                if not is_relevant:
                    print(f"[Pipeline] Skipping irrelevant job (confidence: {confidence:.2f})")
                    continue

                # Get location (use "Unknown" if None to satisfy NOT NULL constraint)
                location = job_data.get('location') or "Unknown"
                if location == "Unknown":
                    print("[Pipeline] ⚠️  No location provided, using 'Unknown'")

                location_matched, reason = matches_location(job_data.get('location'))
                fit_score = calculate_fit_score(job_data['title'], description)
                company = db.query(Company).filter(Company.name == job_data['company']).first()
                if not company:
                    company = Company(
                        name=job_data['company'],
                        domain=extract_domain_from_url(job_data.get('company_url', '')),
                        industry=job_data.get('company_industry'),
                        research_status='pending'
                    )
                    db.add(company)
                    db.flush()
                job = Job(
                    run_id=run_id,
                    company_name=job_data['company'],
                    role_title=job_data['title'],
                    location=location,  # Use sanitized location (never None)
                    location_matched=location_matched,
                    posted_date=job_data.get('date_posted'),
                    source_url=job_data['job_url'],
                    snippet=description[:500] if description else job_data['title'],  # Use title if no description
                    fit_score=fit_score,
                    status='discovered',
                    fetched_text=description,
                    job_source=job_data['site'],
                    job_type=job_data.get('job_type', '')[:50] if job_data.get('job_type') else None,  # Truncate to 50 chars
                    is_remote=job_data.get('is_remote', False),
                    salary_info=job_data.get('salary'),
                    company_url=job_data.get('company_url'),
                )
                db.add(job)
                db.flush()
                jobs_found += 1
                print(f"[Pipeline] ✓ Saved job: {job_data['title'][:50]} at {job_data['company']}")

                # Extract contacts if company URL available (optional - can be disabled)
                if os.getenv('PHASE1_EXTRACT_CONTACTS', 'true').lower() == 'false':
                    continue  # Skip Phase 1 contact extraction

                company_url = job_data.get('company_url')
                if company_url:
                    try:
                        print(f"[Pipeline] Extracting contacts for {job_data['company']}...")
                        from app.extract.emails import (
                            extract_emails,
                            rank_contacts_by_relevance,
                        )
                        from app.fetch.crawler import fetch_page

                        # Fetch company page
                        page_content = await fetch_page(company_url)
                        if page_content:
                            # Extract emails using regex
                            extracted_emails_obj = extract_emails(page_content, company_url)

                            if extracted_emails_obj:
                                # Rank contacts using simple heuristics (no LLM!)
                                ranked_contacts = rank_contacts_by_relevance(extracted_emails_obj)

                                for contact_data in ranked_contacts:
                                    if contact_data['confidence'] < 0.5:
                                        continue
                                    contact = db.query(Contact).filter(Contact.email == contact_data['email']).first()
                                    if not contact:
                                        contact = Contact(
                                            company_id=company.id,
                                            email=contact_data['email'],
                                            name=contact_data.get('name'),
                                            title=contact_data.get('title'),
                                            source_count=1,
                                            status='discovered'
                                        )
                                        db.add(contact)
                                        db.flush()
                                        contacts_found += 1
                                        print(f"[Pipeline] ✓ Contact: {contact_data['email']} (confidence: {contact_data['confidence']:.2f})")
                            else:
                                print("[Pipeline] No emails found on company page")
                        else:
                            print("[Pipeline] Could not fetch company page")
                    except Exception as e:
                        print(f"[Pipeline] Error extracting contacts: {e}")
                else:
                    print("[Pipeline] No company URL available for contact extraction")

        db.commit()
        run.jobs_found = jobs_found
        run.companies_eligible = db.query(Company).count()
        run.status = 'completed'
        run.completed_at = datetime.now()
        db.commit()
        print(f"[Pipeline] Run #{run_id} completed. Found {jobs_found} jobs, {contacts_found} contacts")
        return {"jobs_found": jobs_found, "contacts_found": contacts_found, "message": "Pipeline completed successfully"}
    except Exception as e:
        print(f"[Pipeline] Error in run #{run_id}: {e}")
        import traceback
        traceback.print_exc()
        run.status = 'failed'
        run.error_message = str(e)
        run.completed_at = datetime.now()
        try:
            db.rollback()  # Rollback failed transaction before committing run status
            db.commit()
        except Exception as commit_error:
            print(f"[Pipeline] Failed to save error status: {commit_error}")
        raise

def extract_domain_from_url(url: str) -> str:
    """Extract domain from URL"""
    if not url:
        return ""
    from urllib.parse import urlparse
    try:
        parsed = urlparse(url)
        domain = parsed.netloc or parsed.path
        domain = domain.replace('www.', '')
        return domain
    except Exception:
        return ""

async def research_company_contacts(
    company_name: str,
    db: Session,
    max_contacts: int = 5
) -> Dict[str, Any]:
    """
    Use LangGraph agent with Bedrock to find personal recruiter contacts

    Args:
        company_name: Company to research
        db: Database session
        max_contacts: Max contacts to find

    Returns:
        Dict with personal and fallback contacts
    """

    print(f"[Research] Starting agentic research for {company_name}")

    # Get/create company
    company = db.query(Company).filter(Company.name == company_name).first()
    if not company:
        company = Company(name=company_name, research_status='in_progress')
        db.add(company)
        db.flush()
    else:
        company.research_status = 'in_progress'
        company.research_attempts += 1
        db.commit()

    # Retry logic: max 3 attempts
    max_attempts = 3
    attempt = company.research_attempts

    if attempt > max_attempts:
        print(f"[Research] Max attempts ({max_attempts}) reached, marking as failed")
        company.research_status = 'failed'
        db.commit()

        return {
            'company': company_name,
            'personal_contacts': [],
            'fallback_contacts': [],
            'total_contacts': 0,
            'success': False,
            'error': f'Max retry attempts ({max_attempts}) exceeded'
        }

    try:
        print(f"[Research] Attempt {attempt}/{max_attempts}")

        # Create and run agent
        agent = await _create_research_agent()
        contacts_found = await _run_agent_research(agent, company_name, company, db)

        # Mark complete
        company.research_status = 'completed'
        company.researched_at = datetime.now()
        db.commit()

        # Categorize by email pattern (simple heuristic)
        # Personal: has firstname.lastname or similar patterns
        # Generic: sales@, info@, careers@, etc.
        personal = [c for c in contacts_found if c['email'] and '@' in c['email'] and '.' in c['email'].split('@')[0]]
        fallback = [c for c in contacts_found if c not in personal]

        print(f"[Research] FOUND {len(contacts_found)} contacts")
        print(f"[Research]   {len(personal)} personal, {len(fallback)} fallback")

        return {
            'company': company_name,
            'personal_contacts': personal,
            'fallback_contacts': fallback,
            'total_contacts': len(contacts_found),
            'success': True
        }

    except Exception as e:
        print(f"[Research] ERROR on attempt {attempt}: {e}")

        # Don't mark as failed yet, will retry
        if attempt < max_attempts:
            company.research_status = 'pending'
            print(f"[Research] Will retry (attempts left: {max_attempts - attempt})")
        else:
            company.research_status = 'failed'
            print("[Research] Max attempts reached, marking as failed")

        db.commit()

        return {
            'company': company_name,
            'personal_contacts': [],
            'fallback_contacts': [],
            'total_contacts': 0,
            'success': False,
            'error': str(e),
            'will_retry': attempt < max_attempts
        }

"""
Contact prioritization logic
Priority: 1=highest (personal), 5=lowest (generic)
"""
from typing import Dict


def calculate_contact_priority(email: str, name: str = None, title: str = None, source_type: str = None) -> Dict[str, any]:
    """
    Calculate priority and classification for a contact
    
    Returns:
        {
            'priority': int (1-5),
            'email_type': str (personal, generic, inferred),
            'confidence': float (0.0-1.0)
        }
    """
    
    if not email or '@' not in email:
        return {'priority': 5, 'email_type': 'unknown', 'confidence': 0.1}
    
    local_part = email.split('@')[0].lower()
    
    # Priority 1: Personal emails (firstname.lastname or firstname)
    if '.' in local_part and len(local_part.split('.')) == 2:
        # Looks like firstname.lastname
        parts = local_part.split('.')
        if all(len(p) >= 2 for p in parts):  # Both parts at least 2 chars
            return {
                'priority': 1,
                'email_type': 'personal',
                'confidence': 0.9 if name else 0.7
            }
    
    # Check if name matches email pattern
    if name:
        name_lower = name.lower().replace(' ', '')
        if name_lower in local_part.replace('.', '').replace('_', ''):
            return {
                'priority': 1,
                'email_type': 'personal',
                'confidence': 0.95
            }
    
    # Priority 2: Named recruiter-specific emails
    recruiter_keywords = ['recruit', 'talent', 'hiring', 'hr']
    if any(kw in local_part for kw in recruiter_keywords):
        if name and title:
            return {
                'priority': 2,
                'email_type': 'personal',
                'confidence': 0.8
            }
        else:
            # Generic recruiter email
            return {
                'priority': 4,
                'email_type': 'generic',
                'confidence': 0.5
            }
    
    # Priority 3: Inferred emails (from pattern matching)
    if source_type == 'inferred':
        return {
            'priority': 3,
            'email_type': 'inferred',
            'confidence': 0.6
        }
    
    # Priority 5: Generic/fallback emails
    generic_prefixes = ['careers', 'jobs', 'hr', 'info', 'contact', 'hello', 'support', 'admin']
    if local_part in generic_prefixes:
        return {
            'priority': 5,
            'email_type': 'generic',
            'confidence': 0.9  # High confidence it's generic
        }
    
    # Default: Unknown pattern
    return {
        'priority': 4,
        'email_type': 'unknown',
        'confidence': 0.4
    }


def rank_contacts(contacts: list) -> list:
    """
    Rank contacts by priority and confidence
    Returns sorted list (best first)
    """
    return sorted(contacts, key=lambda c: (c.get('priority', 5), -c.get('confidence', 0)))

@tool
def web_search(query: str, max_results: int = 10) -> List[Dict[str, str]]:

    Think like Google AI: craft queries that will find what you need.

    Args:

    return [
        {
            'title': r['title'],
            'snippet': r['snippet'],
            'url': r['url']
        }
        for r in results
    ]


@tool
async def fetch_and_read_page(url: str) -> Dict[str, Any]:
    """
    Fetch and read content from a URL. Automatically extracts and SAVES any emails found.

    Use this when search results show a promising page.

    Args:
        url: URL to fetch and read

    Returns:
        Dictionary with content, emails found, and save status
    """
    content = await fetch_page(url)

    if not content:
        return {
            'success': False,
            'error': 'Failed to fetch page',
            'emails': [],
            'content': ''
        }

    extracted = extract_emails(content, url)

    # Debug logging
    print(f"[fetch_and_read_page] Extracted {len(extracted)} emails from {url}")
    
    saved_count = 0
    if extracted:
        print(f"[fetch_and_read_page] Emails found: {[e.email for e in extracted]}")
        
        # AUTO-SAVE to database
        try:
            from app.models.contact import Contact, Evidence
            
            ctx = get_tool_context()
            company = ctx.get('company')
            db = ctx.get('db')
            
            if company and db:
                for e in extracted:
                    # Skip duplicates
                    existing = db.query(Contact).filter(Contact.email == e.email).first()
                    if existing:
                        continue
                    
                    # Create contact
                    contact = Contact(
                        company_id=company.id,
                        email=e.email,
                        name=e.name,
                        title=e.title,
                        inferred=False,
                        status='discovered'
                    )
                    db.add(contact)
                    db.flush()
                    
                    # Add evidence
                    evidence = Evidence(
                        contact_id=contact.id,

@router.post("/run")
async def trigger_run(
    db: Session = Depends(get_db),
    authorized: bool = Depends(verify_api_key)
):
    """
    Trigger GitHub Actions workflow to run the pipeline
    
    This endpoint:
    1. Creates a Run record in the database
    2. Triggers GitHub Actions workflow via API
    3. Returns immediately (async execution)
    """
    
    try:
        # Get GitHub config from environment
        github_token = os.getenv('GITHUB_TOKEN')
        github_repo = os.getenv('GITHUB_REPO')  # e.g., "username/repo-name"
        
        if not github_token or not github_repo:
            raise HTTPException(
                status_code=500,
                detail="GitHub integration not configured. Set GITHUB_TOKEN and GITHUB_REPO."
            )
        
        # Create new run record (will be picked up by workflow)
        run = Run(
            started_at=datetime.now(),
            status='queued'
        )
        db.add(run)
        db.commit()
        db.refresh(run)
        
        # Trigger GitHub Actions workflow
        workflow_url = f"https://api.github.com/repos/{github_repo}/actions/workflows/daily-run.yml/dispatches"
        
        headers = {
            'Authorization': f'Bearer {github_token}',
            'Accept': 'application/vnd.github+json',
            'X-GitHub-Api-Version': '2022-11-28'
        }
        
        payload = {
            'ref': 'main',  # or 'master' depending on your default branch
            'inputs': {
                'max_jobs': '50'
            }
        }
        
        async with httpx.AsyncClient() as client:
            response = await client.post(
                workflow_url,
                headers=headers,
                json=payload,
                timeout=10.0
            )
        
        if response.status_code == 204:
            # Success - workflow triggered
            return {
                "success": True,
                "message": "Pipeline queued via GitHub Actions",
                "run_id": run.id,
                "status": "queued"
            }
        else:
            # Failed to trigger
            run.status = 'failed'
            run.error_message = f"GitHub API error: {response.text}"
            db.commit()
            
            raise HTTPException(
                status_code=500,
                detail=f"Failed to trigger workflow: {response.text}"
            )
    
    except HTTPException:
        raise
    except Exception as e:
        raise HTTPException(
            status_code=500,
            detail=f"Error triggering pipeline: {str(e)}"
        )

        }

        # Rollback: Delete associated jobs and contacts
        from app.models import Job
        
        # Delete jobs from this run
        db.query(Job).filter(Job.run_id == run_id).delete()
        
        # Mark run as cancelled
        run.status = 'cancelled'
        run.completed_at = datetime.now()
        run.error_message = "Cancelled by user (work discarded)"
        run.jobs_found = 0

@router.get("/open/{token}")
async def track_open(token: str, request: Request):
    """
    Track email open event
    Returns 1x1 transparent GIF
    """
    db = next(get_db())
    
    try:
        # Find contact by tracking token
        contact = db.query(Contact).filter(
            Contact.tracking_token == token
        ).first()
        
        if not contact:
            # Return pixel even if token not found (don't leak info)
            return StreamingResponse(io.BytesIO(TRACKING_PIXEL), media_type="image/gif")
        
        # Get user agent and IP
        user_agent = request.headers.get('user-agent', '')
        client_ip = request.client.host if request.client else ''
        ip_hash = hash_ip(client_ip) if client_ip else None
        
        # Detect machine open (email scanners, etc.)
        is_machine = detect_machine_open(user_agent)
        
        # Find associated outreach (most recent sent to this contact)
        from app.models.outreach import Outreach
        outreach = db.query(Outreach).filter(
            Outreach.contact_id == contact.id
        ).order_by(Outreach.sent_at.desc()).first()
        
        # Record open event
        open_event = Open(
            contact_id=contact.id,
            outreach_id=outreach.id if outreach else None,
            opened_at=datetime.now(),
            user_agent=user_agent[:500] if user_agent else None,
            ip_hash=ip_hash,
            is_machine_open=is_machine
        )
        db.add(open_event)
        db.commit()
        
        print(f"[Tracking] Open tracked for contact {contact.id} (machine={is_machine})")
        
    except Exception as e:
        print(f"[Tracking] Error: {e}")
    finally:
        db.close()
    
    # Always return tracking pixel
    return StreamingResponse(io.BytesIO(TRACKING_PIXEL), media_type="image/gif")

def detect_machine_open(user_agent: str) -> bool:
    """Detect if open is from email scanner/machine"""
    if not user_agent:
        return True
    
    ua_lower = user_agent.lower()
    
    # Known email scanners
    machine_patterns = [
        'bot', 'spider', 'crawler', 'scan', 'check',
        'proofpoint', 'mimecast', 'barracuda', 'ironport',
        'google image proxy', 'google-image-proxy', 'microsoft', 'outlook-android-sync'
    ]
    
    for pattern in machine_patterns:
        if pattern in ua_lower:
            return True
    
    return False

"""
JobSpy integration for direct job scraping from LinkedIn, Naukri, Indeed, Glassdoor, ZipRecruiter
"""
import os
import time
from typing import Any, Dict, List, Optional

import pandas as pd


class JobSpyScraper:
    """
    Scrape jobs from multiple sites with intelligent rate limiting
    """

    # Rate limit settings (requests per minute) - conservative for reliability
    RATE_LIMITS = {
        'linkedin': 30,      # 1 req per 2 sec
        'naukri': 30,        # 1 req per 2 sec
        'indeed': 120,       # No rate limit (best scraper)
        'glassdoor': 20,     # 1 req per 3 sec (aggressive blocking)
        'ziprecruiter': 30,  # 1 req per 2 sec
    }

    def __init__(self):
        self.last_request_time = {}
        self.retry_wait_time = 60  # Initial wait on 429

    def scrape_jobs(
        self,
        search_term: str,
        location: str,
        sites: List[str] = ["linkedin", "naukri", "indeed", "glassdoor"],
        results_wanted: int = 20,
        hours_old: int = 168,  # 7 days
        country: str = "India"
    ) -> List[Dict[str, Any]]:
        """
        Scrape jobs from multiple sites with rate limiting

        Args:
            search_term: Job search query (e.g., "GenAI engineer")
            location: Job location
            sites: List of sites to scrape
            results_wanted: Max results per site
            hours_old: Filter jobs posted within N hours (default: 7 days)
            country: Country for Indeed

        Returns:
            List of job dictionaries
        """
        from jobspy import scrape_jobs as jobspy_scrape

        print(f"[JobSpy] Scraping {', '.join(sites)}")
        print(f"[JobSpy] Query: '{search_term}' in {location}")
        print(f"[JobSpy] Results wanted: {results_wanted} per site, within {hours_old}h")

        # Apply rate limiting before request
        self._wait_for_rate_limit(sites)

        # Try scraping with retries
        for attempt in range(3):
            try:
                jobs_df = jobspy_scrape(
                    site_name=sites,
                    search_term=search_term,
                    location=location,
                    results_wanted=results_wanted,
                    hours_old=hours_old,
                    country_indeed=country,
                )

                if jobs_df is None or jobs_df.empty:
                    print("[JobSpy] No jobs found")
                    return []

                print(f"[JobSpy] âœ“ Found {len(jobs_df)} jobs")

                # Convert to list of dicts
                jobs = self._convert_to_jobs(jobs_df)
                return jobs

            except Exception as e:
                error_str = str(e)

                # Check for rate limiting (429)
                if '429' in error_str or 'rate limit' in error_str.lower():
                    wait_time = self.retry_wait_time * (2 ** attempt)  # Exponential backoff
                    print(f"[JobSpy] âš  Rate limited (429). Waiting {wait_time}s before retry {attempt+1}/3")
                    time.sleep(wait_time)
                    continue

                # Other errors
                print(f"[JobSpy] âœ— Error on attempt {attempt+1}/3: {e}")

                if attempt < 2:
                    time.sleep(10)  # Short wait before retry
                    continue
                else:
                    print("[JobSpy] Max retries reached, skipping this search")
                    return []

        return []

    def _wait_for_rate_limit(self, sites: List[str]):
        """Apply rate limiting delays"""
        # Find the most restrictive rate limit
        min_delay = 0
        for site in sites:
            if site in self.RATE_LIMITS:
                rpm = self.RATE_LIMITS[site]
                delay = 60.0 / rpm  # Seconds between requests
                min_delay = max(min_delay, delay)

        # Check if we need to wait
        for site in sites:
            if site in self.last_request_time:
                elapsed = time.time() - self.last_request_time[site]
                if elapsed < min_delay:
                    wait = min_delay - elapsed
                    print(f"[JobSpy] Rate limiting: waiting {wait:.1f}s for {site}")
                    time.sleep(wait)

        # Update last request time
        current_time = time.time()
        for site in sites:
            self.last_request_time[site] = current_time

    def _convert_to_jobs(self, jobs_df: pd.DataFrame) -> List[Dict[str, Any]]:
        """Convert DataFrame to list of job dicts"""
        jobs = []

        for _, row in jobs_df.iterrows():
            # Format salary
            salary = self._format_salary(row)

            # Convert date_posted to date object
            date_posted = None
            if pd.notna(row.get('date_posted')):
                try:
                    date_posted = pd.to_datetime(row.get('date_posted')).date()
                except Exception:
                    pass

            job = {
                'site': row.get('site'),
                'title': row.get('title'),
                'company': row.get('company'),
                'location': row.get('location'),
                'date_posted': date_posted,
                'job_url': row.get('job_url'),
                'job_url_direct': row.get('job_url_direct'),
                'description': row.get('description'),
                'job_type': row.get('job_type'),  # fulltime, parttime, etc.
                'salary': salary,
                'is_remote': bool(row.get('is_remote')) if pd.notna(row.get('is_remote')) else False,
                'company_url': row.get('company_url'),
                'company_industry': row.get('company_industry'),
            }

            jobs.append(job)

        return jobs

    def _format_salary(self, row) -> Optional[str]:
        """Format salary info from row"""
        try:
            if pd.isna(row.get('min_amount')) and pd.isna(row.get('max_amount')):
                return None

            min_amt = row.get('min_amount')
            max_amt = row.get('max_amount')
            currency = row.get('currency', 'â‚¹')
            interval = row.get('interval', 'yearly')

            if pd.notna(min_amt) and pd.notna(max_amt):
                return f"{currency}{int(min_amt)}-{int(max_amt)}/{interval}"
            elif pd.notna(min_amt):
                return f"{currency}{int(min_amt)}+/{interval}"
            elif pd.notna(max_amt):
                return f"Up to {currency}{int(max_amt)}/{interval}"

            return None
        except Exception:
            return None


def build_jobspy_searches() -> List[Dict[str, Any]]:
    """
    Build JobSpy search configurations for Remote + Chennai

    Returns:

import requests
from bs4 import BeautifulSoup
from typing import List

class SearchResult:
    """Search result data class"""
    def __init__(self, title: str, url: str, snippet: str, date: str = None):
        self.title = title
        self.url = url
        self.snippet = snippet
        self.date = date
    
    def to_dict(self):
        return {
            'title': self.title,
            'url': self.url,
            'snippet': self.snippet,
            'date': self.date
        }

class SearchError(Exception):
    """Search provider error"""
    def __init__(self, message: str, provider: str, is_quota_error: bool = False):
        super().__init__(message)
        self.provider = provider
        self.is_quota_error = is_quota_error

class DuckDuckGoSearch:
    """DuckDuckGo HTML search (free, unlimited)"""
    
    name = 'duckduckgo'
    
    def search(self, query: str, max_results: int = 10, date_filter: str = None) -> List[SearchResult]:
        """
        Search DuckDuckGo
        date_filter: 'd' for day, 'w' for week, 'm' for month
        """
        url = f"https://html.duckduckgo.com/html/?q={requests.utils.quote(query)}"
        
        if date_filter:
            url += f"&df={date_filter}"
        
        headers = {
            'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36'
        }
        
        try:
            response = requests.get(url, headers=headers, timeout=10)
            response.raise_for_status()
            
            soup = BeautifulSoup(response.text, 'html.parser')
            results = []
            
            for result_div in soup.find_all('div', class_='result', limit=max_results):
                title_elem = result_div.find('a', class_='result__a')
                snippet_elem = result_div.find('a', class_='result__snippet')
                
                if not title_elem:
                    continue
                
                title = title_elem.get_text(strip=True)
                result_url = title_elem.get('href', '')
                snippet = snippet_elem.get_text(strip=True) if snippet_elem else ''
                
                # Extract actual URL from DuckDuckGo redirect
                if '//duckduckgo.com/l/?' in result_url:
                    import re
                    match = re.search(r'uddg=([^&]+)', result_url)
                    if match:
                        result_url = requests.utils.unquote(match.group(1))
                
                if title and result_url and snippet:
                    results.append(SearchResult(title, result_url, snippet))
                
                if len(results) >= max_results:
                    break
            
            return results
            
        except Exception as e:
            raise SearchError(f"DuckDuckGo search failed: {str(e)}", 'duckduckgo')

class SerperSearch:
    """Serper.dev Google Search API"""
    
    name = 'serper'
    
    def __init__(self, api_key: str):
        self.api_key = api_key
    
    def search(self, query: str, max_results: int = 10, date_filter: str = None) -> List[SearchResult]:
        """Search using Serper API"""
        
        payload = {
            'q': query,
            'num': max_results,
            'gl': 'in'  # India
        }
        
        if date_filter:
            # Map to Google's date filter format
            mapping = {'d': 'qdr:d', 'w': 'qdr:w', 'm': 'qdr:m'}
            payload['tbs'] = mapping.get(date_filter, 'qdr:d2')
        
        headers = {
            'X-API-KEY': self.api_key,
            'Content-Type': 'application/json'
        }
        
        try:
            response = requests.post(
                'https://google.serper.dev/search',
                json=payload,
                headers=headers,
                timeout=10
            )
            response.raise_for_status()
            
            data = response.json()
            results = []
            
            for item in data.get('organic', []):
                results.append(SearchResult(
                    title=item.get('title', ''),
                    url=item.get('link', ''),
                    snippet=item.get('snippet', ''),
                    date=item.get('date')
                ))
            
            return results
            
        except requests.exceptions.HTTPError as e:
            is_quota = e.response.status_code == 429
            raise SearchError(f"Serper search failed: {str(e)}", 'serper', is_quota)
        except Exception as e:
            raise SearchError(f"Serper search failed: {str(e)}", 'serper')


class FirecrawlSearch:
    """Firecrawl Search API"""
    
    name = 'firecrawl'
    
    def __init__(self, api_key: str):
        self.api_key = api_key
    
    def search(self, query: str, max_results: int = 10, date_filter: str = None) -> List[SearchResult]:

            'query': query,
            'limit': max_results
        }
        
        headers = {
            'Authorization': f'Bearer {self.api_key}',
            'Content-Type': 'application/json'
        }
        
        try:

                'https://api.firecrawl.dev/v1/search',

                headers=headers,
                timeout=15
            )
            response.raise_for_status()
            
            data = response.json()
            results = []
            
            # Firecrawl returns results in 'data' array

                results.append(SearchResult(
                    title=item.get('title', ''),
                    url=item.get('url', ''),
                    snippet=item.get('description', ''),
                    date=None
                ))
            
            return results
            
        except requests.exceptions.HTTPError as e:
            is_quota = e.response.status_code == 429
            raise SearchError(f"Firecrawl search failed: {str(e)}", 'firecrawl', is_quota)
        except Exception as e:
            raise SearchError(f"Firecrawl search failed: {str(e)}", 'firecrawl')

"""
Search router with caching and failover
"""
import hashlib
import json
import os
from datetime import datetime, timedelta
from typing import List, Optional
from sqlalchemy.orm import Session
from app.search.providers import (
    DuckDuckGoSearch, SerperSearch, FirecrawlSearch, SearchResult, SearchError
)
from app.models.outreach import CacheSearch

class SearchRouter:
    """Routes searches through providers with caching and failover"""
    
    def __init__(self, db: Session, cache_enabled: bool = True):
        self.db = db
        self.cache_enabled = cache_enabled
        self.providers = []
        
        # Get configuration from environment
        search_provider = os.getenv('SEARCH_PROVIDER', 'serper').lower()  # serper or firecrawl
        serper_key = os.getenv('SERPER_API_KEY')
        firecrawl_key = os.getenv('FIRECRAWL_API_KEY')
        
        # Setup providers based on configuration
        if search_provider == 'firecrawl':
            # Firecrawl as primary, Serper as fallback
            if firecrawl_key:
                self.providers.append(FirecrawlSearch(firecrawl_key))
                print("[SearchRouter] Primary: Firecrawl")
            if serper_key:
                self.providers.append(SerperSearch(serper_key))
                print("[SearchRouter] Fallback: Serper")
        else:
            # Serper as primary (default), Firecrawl as fallback
            if serper_key:
                self.providers.append(SerperSearch(serper_key))
                print("[SearchRouter] Primary: Serper")
            if firecrawl_key:
                self.providers.append(FirecrawlSearch(firecrawl_key))
                print("[SearchRouter] Fallback: Firecrawl")
        
        # Always add DuckDuckGo as final fallback (free, unlimited)
        self.providers.append(DuckDuckGoSearch())
        print("[SearchRouter] Final fallback: DuckDuckGo")

    
    def search(self, query: str, max_results: int = 10, date_filter: str = None) -> List[SearchResult]:
        """
        Search with caching and failover
        date_filter: 'd' for day, 'w' for week, 'm' for month
        """
        # Check cache first
        if self.cache_enabled:
            cached = self._get_from_cache(query, max_results, date_filter)
            if cached:
                print(f"[SearchRouter] Cache hit for: {query[:50]}...")
                return cached
        
        # Try each provider with failover
        errors = []
        
        for provider in self.providers:
            try:
                print(f"[SearchRouter] Trying {provider.name} for: {query[:50]}...")
                results = provider.search(query, max_results, date_filter)
                
                # Cache successful results
                if self.cache_enabled and results:
                    self._save_to_cache(query, max_results, date_filter, results)
                
                print(f"[SearchRouter] {provider.name} returned {len(results)} results")
                return results
                
            except SearchError as e:
                errors.append(e)
                print(f"[SearchRouter] {provider.name} failed: {e}")
                
                # Skip to next provider on quota error
                if e.is_quota_error:
                    continue
                
                # Also try next provider on other errors
                continue
        
        # All providers failed
        error_msg = "All search providers failed:\n" + "\n".join(
            f"{e.provider}: {str(e)}" for e in errors
        )
        raise Exception(error_msg)
    
    def _get_from_cache(self, query: str, max_results: int, date_filter: Optional[str]) -> Optional[List[SearchResult]]:
        """Get results from cache"""
        query_hash = self._hash_query(query, max_results, date_filter)
        
        cache_entry = self.db.query(CacheSearch).filter(
            CacheSearch.query_hash == query_hash,
            CacheSearch.expires_at > datetime.now()
        ).first()
        
        if cache_entry:
            results_data = json.loads(cache_entry.results)
            return [SearchResult(**r) for r in results_data]
        
        return None
    
    def _save_to_cache(self, query: str, max_results: int, date_filter: Optional[str], results: List[SearchResult]):
        """Save results to cache"""
        query_hash = self._hash_query(query, max_results, date_filter)
        expires_at = datetime.now() + timedelta(hours=24)
        
        results_json = json.dumps([r.to_dict() for r in results])
        
        cache_entry = self.db.query(CacheSearch).filter(
            CacheSearch.query_hash == query_hash
        ).first()
        
        if cache_entry:
            cache_entry.results = results_json
            cache_entry.expires_at = expires_at
        else:
            cache_entry = CacheSearch(
                query_hash=query_hash,
                results=results_json,
                expires_at=expires_at
            )
            self.db.add(cache_entry)
        
        self.db.commit()
    
    def _hash_query(self, query: str, max_results: int, date_filter: Optional[str]) -> str:
        """Hash query parameters for cache key"""
        key = json.dumps({
            'query': query,
            'max_results': max_results,
            'date_filter': date_filter
        })
        return hashlib.sha256(key.encode()).hexdigest()

async def run_phase2_research(run_id: int, db: Session, limit: int = None):
    """Phase 2: Research Contacts with Parallel Processing"""
    from app.research.agent import research_company_contacts
    
    # Get configuration from environment
    if limit is None:
        limit = int(os.getenv('RESEARCH_COMPANY_LIMIT', '50'))
    workers = int(os.getenv('RESEARCH_PARALLEL_WORKERS', '3'))
    
    print("\n" + "=" * 60)
    print(f"PHASE 2: CONTACT RESEARCH (PARALLEL - {workers} workers)")
    print("=" * 60)
    
    run = db.query(Run).filter(Run.id == run_id).first()
    run.phase = 'research'
    db.commit()
    
    companies = db.query(Company).filter(
        Company.research_status == 'pending'
    ).limit(limit).all()
    
    print(f"Researching {len(companies)} companies with {workers} concurrent workers...")
    print("(Rate limited to prevent system overload)")
    
    start = time.time()
    
    # Semaphore to limit concurrent research tasks
    # GitHub Actions: 2 cores, 7GB RAM
    # Configurable via RESEARCH_PARALLEL_WORKERS (2-5 recommended)
    semaphore = asyncio.Semaphore(workers)
    
    async def research_with_limit(company, index):
        """Research company with semaphore rate limiting"""
        async with semaphore:
            print(f"\n[{index}/{len(companies)}] {company.name}")
            try:
                result = await research_company_contacts(company.name, db)
                
                if result['success']:
                    print(f"  âœ“ Found {result['total_contacts']} contacts ({len(result['personal_contacts'])} personal)")
                else:
                    print(f"  âœ— Failed: {result.get('error', 'Unknown error')}")
                
                return result
            except Exception as e:
                print(f"  âœ— Error: {e}")
                return {
                    'company': company.name,
                    'success': False,
                    'error': str(e),
                    'personal_contacts': [],
                    'fallback_contacts': [],
                    'total_contacts': 0
                }
    
    # Create tasks for all companies
    tasks = [research_with_limit(company, i+1) for i, company in enumerate(companies)]
    
    # Execute with asyncio.gather (handles exceptions gracefully)
    results = await asyncio.gather(*tasks, return_exceptions=True)
    
    # Filter out exceptions and count results
    valid_results = [r for r in results if isinstance(r, dict)]
    
    elapsed = time.time() - start
    
    total_contacts = sum(r.get('total_contacts', 0) for r in valid_results)
    personal_contacts = sum(len(r.get('personal_contacts', [])) for r in valid_results)
    successful = sum(1 for r in valid_results if r.get('success', False))
    
    run.companies_researched = len(companies)
    run.contacts_found = total_contacts
    run.personal_contacts_found = personal_contacts
    db.commit()
    
    print(f"\nâœ“ Completed in {elapsed:.1f}s")
    print(f"  Companies researched: {len(companies)} ({successful} successful)")

"""

from pathlib import Path
from typing import Any, Dict

import yaml

_cache: Dict[str, Any] = {}

def load_config(filename: str) -> Dict[str, Any]:
    """Load YAML config file with caching"""
    if filename in _cache:
        return _cache[filename]

    config_path = Path(__file__).parent.parent / 'config' / filename
    with open(config_path, 'r') as f:
        config = yaml.safe_load(f)

    _cache[filename] = config
    return config

def get_keywords() -> Dict[str, Any]:
    """Get keywords configuration"""
    return load_config('keywords.yaml')

def get_budgets() -> Dict[str, Any]:
    """Get budgets configuration"""
    return load_config('budgets.yaml')

def get_profile() -> Dict[str, Any]:
    """Get user profile configuration"""
    return load_config('profile.yaml')

def matches_location(location: str) -> tuple[bool, str]:
    """
    Check if location matches Chennai or Remote (India)
    Returns: (matches, reason if not matched)
    """
    if not location:
        return False, "No location provided"

    keywords = get_keywords()
    location_lower = location.lower()

    # Check exclusions first
    for exclude in keywords['locations']['exclude_patterns']:
        if exclude.lower() in location_lower:
            return (False, f"Excluded: {exclude}")

    # Check Chennai
    for chennai in keywords['locations']['chennai']:
        if chennai.lower() in location_lower:
            return (True, "")

    # Check remote
    for remote in keywords['locations']['remote']:
        if remote.lower() in location_lower:
            return (True, "")

    return (False, "Not Chennai or Remote (India)")

def calculate_fit_score(job_title: str, job_text: str = "") -> int:
    """
    Calculate job fit score 0-100
    """
    if not job_title:
        return 0

    profile = get_profile()
    keywords = get_keywords()

    title_lower = job_title.lower()
    text_lower = (job_text or "").lower()
    combined = f"{title_lower} {text_lower}"

    score = 0

    # Primary keywords (5 points each, max 40)
    for keyword in keywords['keywords']['primary'][:8]:
        if keyword.lower() in combined:
            score += 5
            if score >= 40:
                break

    # Secondary keywords (2 points each, max 20)
    secondary_score = 0
    for keyword in keywords['keywords']['secondary'][:10]:
        if keyword.lower() in combined:
            secondary_score += 2
            if secondary_score >= 20:
                break
    score += secondary_score

    # User skills (3 points each, max 30)
    skill_score = 0
    for skill in profile['skills'][:10]:
        if skill.lower() in combined:
            skill_score += 3
            if skill_score >= 30:
                break
    score += skill_score

    # Check exclusions (-20 each)
    for exclusion in keywords['exclusions']:
        if exclusion.lower() in combined:
            score -= 20

    return max(0, min(100, score))

def select_resume_variant(job_title: str) -> str:
    """
    Select appropriate resume variant based on job title
    """
    profile = get_profile()
    title_lower = job_title.lower()

    mapping = profile.get('resume_mapping', {})
    for variant, keywords_list in mapping.items():
        for keyword in keywords_list:
            if keyword.lower() in title_lower:
                return profile['resume_variants'].get(variant,
                                                       profile['resume_variants']['default'])

    return profile['resume_variants']['default']

"""
Database connection and utilities
"""
import os
import hashlib
from sqlalchemy import create_engine
from sqlalchemy.orm import sessionmaker, declarative_base
from dotenv import load_dotenv

load_dotenv()

DATABASE_URL = os.getenv('DATABASE_URL')
if not DATABASE_URL:
    raise ValueError('DATABASE_URL must be set in environment variables')

# Fix DATABASE_URL format for SQLAlchemy 2.x (postgres:// -> postgresql://)
if DATABASE_URL.startswith('postgres://'):
    DATABASE_URL = DATABASE_URL.replace('postgres://', 'postgresql://', 1)

engine = create_engine(DATABASE_URL, pool_size=10, max_overflow=20)
SessionLocal = sessionmaker(autocommit=False, autoflush=False, bind=engine)
Base = declarative_base()

def get_db():
    """Get database session"""
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()

def hash_email(email: str) -> str:
    """Hash email for privacy (suppression list)"""
    return hashlib.sha256(email.lower().encode()).hexdigest()

def hash_ip(ip: str) -> str:
    """Hash IP address with salt"""
    salt = os.getenv('IP_HASH_SALT', 'default-salt-change-me')
    return hashlib.sha256((ip + salt).encode()).hexdigest()
